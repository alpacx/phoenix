cmake_minimum_required(VERSION 3.16)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/platforms/esp32/main" "${CMAKE_CURRENT_LIST_DIR}/shared/phoenix_core")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(phoenix)

idf_component_get_property(lwip_lib lwip COMPONENT_LIB)
target_include_directories(${lwip_lib} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/platforms/esp32/main")
target_compile_definitions(${lwip_lib} PRIVATE ESP_IDF_LWIP_HOOK_FILENAME="lwip_hooks.h")

# Override only the reviewed 2.0.1 network translation unit; keep registry sources untouched.
set(tusb_net_original "${CMAKE_CURRENT_LIST_DIR}/managed_components/espressif__esp_tinyusb/tinyusb_net.c")
file(SHA256 "${tusb_net_original}" tusb_net_hash)
if(NOT tusb_net_hash STREQUAL "bbc8a8e59e8e41d7d4cdd5562194059946668237f791ae437f36bcea72d0e063")
    message(FATAL_ERROR "esp_tinyusb network source changed: review the local override")
endif()
idf_component_get_property(tusb_wrapper espressif__esp_tinyusb COMPONENT_LIB)
get_target_property(tusb_sources ${tusb_wrapper} SOURCES)
list(FILTER tusb_sources EXCLUDE REGEX "(^|/)tinyusb_net\\.c$")
list(APPEND tusb_sources "${CMAKE_CURRENT_LIST_DIR}/components_overrides/esp_tinyusb/tinyusb_net.c")
set_property(TARGET ${tusb_wrapper} PROPERTY SOURCES "${tusb_sources}")
